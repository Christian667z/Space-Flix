— SUPABASE SQL EDITOR

Script SQL pour Reprise & Authentification

Copiez ce script et collez-le dans Supabase > SQL Editor > New Query > Run pour configurer la base de données de reprise de lecture, profils, favoris et historique. 

-- Table Reprise de lecture & Historique
CREATE TABLE IF NOT EXISTS public.user_history (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
    media_id TEXT NOT NULL,
    media_title VARCHAR(255),
    media_type VARCHAR(20) DEFAULT 'movie',
    poster_path TEXT,
    backdrop_path TEXT,
    season INTEGER DEFAULT 1,
    episode INTEGER DEFAULT 1,
    progress_seconds NUMERIC DEFAULT 0,
    duration_seconds NUMERIC DEFAULT 0,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
    UNIQUE(user_id, media_id, season, episode)
);

-- Table Favoris ("Ma Liste")
CREATE TABLE IF NOT EXISTS public.favorites (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
    media_id TEXT NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
    UNIQUE(user_id, media_id)
);
